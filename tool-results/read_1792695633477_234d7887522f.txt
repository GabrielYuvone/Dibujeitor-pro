   277→// ---------------------------------------------------------------------------
   278→// Tipo para entradas del stack de undo/redo
   279→// ---------------------------------------------------------------------------
   280→// Dos tipos de entrada:
   281→//  - "drawing": edición in-place de un drawing (pincel, goma, etc.)
   282→//  - "project": cambio estructural (agregar capa, mover cell, extender
   283→//    cell, etc.). Para estos, capturamos una referencia al project
   284→//    PRE-mutación. Como los proyectos son inmutables (cada mutación
   285→//    produce un nuevo objeto), la referencia capturada sigue apuntando
   286→//    al estado anterior. La memoria es eficiente porque los drawings
   287→//    compartidos entre snapshots se referencian, no se duplican.
   288→
   289→type HistoryEntry =
   290→  | { type: "drawing"; drawingId: ID; prevDataUrl: string; newDataUrl: string }
   291→  | { type: "project"; prevProject: AnimationProject };
   292→
   293→// ---------------------------------------------------------------------------
   294→// Tipo para el placement de imagen (transformación interactiva)
   295→// ---------------------------------------------------------------------------
   296→
   297→interface ImagePlacement {
   298→  dataUrl: string;        // imagen original (data URL)
   299→  x: number;               // posición x en coords del canvas (esquina sup-izq)
   300→  y: number;               // posición y
   301→  width: number;           // ancho actual (preserva aspect ratio)
   302→  height: number;          // alto actual
   303→  nativeW: number;         // ancho original de la imagen (px)
   304→  nativeH: number;         // alto original (px)
   305→}

   311→function updateProject(
   312→  state: AppState,
   313→  updater: (p: AnimationProject) => AnimationProject
   314→): Partial<AppState> {
   315→  if (!state.project) return {};
   316→  const newProject = updater({ ...state.project, dirty: true, updatedAt: Date.now() });
   317→  return { project: newProject };
   318→}

   752→  commitLayerDraftAsKeyframe: (layerId, frame) => {
   753→    // Captura el transform ACTUAL (interpolated + draft) como keyframe
   754→    // en el frame actual y resetea el draft a identidad. Así el usuario
   755→    // puede arrastrar libremente sin crear keyframes automáticamente y
   756→    // commitear solo cuando lo decide.
   757→    const state = get();

   761→    const draft = state.layerDrafts[layerId];
   762→    // interp = transform interpolado en este frame (de los kfs existentes)
   763→    const interp = interpolateLayerTransform(layer, frame);
   764→    // final = interp + draft (suma en offset y rotation, multiplica scale)
   765→    const finalTransform = {
   766→      offsetX: interp.offsetX + (draft?.offsetX ?? 0),
   767→      offsetY: interp.offsetY + (draft?.offsetY ?? 0),
   768→      scale: interp.scale * (draft?.scale ?? 1),
   769→      rotation: interp.rotation + (draft?.rotation ?? 0),
   770→    };
   771→    // Push snapshot ANTES de mutar el project (para que sea deshacerable)
   772→    get().pushProjectSnapshot();
   773→    set(updateProject(state, (p) => ({
   774→      ...p,
   775→      layers: p.layers.map((l) => {
   776→        if (l.id !== layerId) return l;
   777→        const kfs = [...(l.transformKeyframes ?? [])];
   778→        const idx = kfs.findIndex((k) => k.frame === frame);
   779→        const newKf = { frame, ...finalTransform };
   780→        if (idx >= 0) kfs[idx] = newKf;
   781→        else {
   782→          kfs.push(newKf);
   783→          kfs.sort((a, b) => a.frame - b.frame);
   784→        }
   785→        return { ...l, transformKeyframes: kfs };
   786→      }),
   787→    })));
   788→    // Resetear el draft — el offset ahora está "bakeado" en el keyframe.
   789→    set((s) => {
   790→      const next = { ...s.layerDrafts };
   791→      delete next[layerId];
   792→      return { layerDrafts: next };
   793→    });

   800→  selectCell: (layerId, cellId, modifiers) => {
   801→    const state = get();
   802→    if (!state.project) return;
   803→    const layer = state.project.layers.find((l) => l.id === layerId);
   804→    if (!layer) return;
   805→    const clickedCell = layer.cells.find((c) => c.id === cellId);
   806→    if (!clickedCell) return;
   807→
   808→    if (modifiers.shift) {
   809→      // Shift+click: seleccionar rango desde el último cell seleccionado
   810→      // (en la misma capa) hasta el cell clickeado. Si no hay selección
   811→      // previa en esta capa, el rango arranca desde el cell clickeado.
   812→      const lastSelectedInLayer = layer.cells.find((c) =>
   813→        state.selectedCellIds.has(c.id)
   814→      );
   815→      const anchor = lastSelectedInLayer ?? clickedCell;
   816→      const startFrame = Math.min(anchor.startFrame, clickedCell.startFrame);
   817→      const endFrame = Math.max(anchor.startFrame, clickedCell.startFrame);
   818→      const range = layer.cells.filter(
   819→        (c) => c.startFrame >= startFrame && c.startFrame <= endFrame
   820→      );
   821→      const newSet = new Set(state.selectedCellIds);
   822→      range.forEach((c) => newSet.add(c.id));
   823→      set({ selectedCellIds: newSet });
   824→    } else if (modifiers.ctrl) {
   825→      // Ctrl/Cmd+click: toggle individual
   826→      const newSet = new Set(state.selectedCellIds);
   827→      if (newSet.has(cellId)) newSet.delete(cellId);
   828→      else newSet.add(cellId);
   829→      set({ selectedCellIds: newSet });
   830→    } else {
   831→      // Click normal: selección única
   832→      set({ selectedCellIds: new Set([cellId]) });
   833→    }
   834→  },

   878→  ensureDrawingBounds: async (drawingId, x, y, margin = 100) => {
   879→    const state = get();
   880→    if (!state.project) return { offsetX: 0, offsetY: 0 };
   881→    const drawing = state.project.drawings[drawingId];
   882→    if (!drawing) return { offsetX: 0, offsetY: 0 };
   883→
   884→    // Calcular cuánto hay que expandir en cada dirección para que (x, y)
   885→    // quede dentro del bitmap, con un margen para no expandir en cada click.
   886→    const expandLeft = x < 0 ? Math.ceil(-x + margin) : 0;
   887→    const expandTop = y < 0 ? Math.ceil(-y + margin) : 0;
   888→    const expandRight = x >= drawing.width ? Math.ceil(x - drawing.width + margin) : 0;
   889→    const expandBottom = y >= drawing.height ? Math.ceil(y - drawing.height + margin) : 0;
   890→
   891→    if (expandLeft === 0 && expandTop === 0 && expandRight === 0 && expandBottom === 0) {
   892→      return { offsetX: 0, offsetY: 0 };
   893→    }
   894→
   895→    // El origen del bitmap se mueve: lo que era (0, 0) ahora está en
   896→    // (expandLeft, expandTop) del nuevo bitmap. Esto es lo que el caller
   897→    // necesita para ajustar el draft (restar expandLeft/Top) y el click
   898→    // (sumar expandLeft/Top).
   899→    const offsetX = expandLeft;
   900→    const offsetY = expandTop;
   901→    const newW = drawing.width + expandLeft + expandRight;
   902→    const newH = drawing.height + expandTop + expandBottom;
   903→
   904→    // Crear nuevo canvas con las nuevas dimensiones
   905→    const newCanvas = document.createElement("canvas");
   906→    newCanvas.width = newW;
   907→    newCanvas.height = newH;
   908→    const newCtx = newCanvas.getContext("2d");
   909→    if (!newCtx) return { offsetX: 0, offsetY: 0 };
   910→
   911→    // Cargar la imagen actual y dibujarla en (offsetX, offsetY) del nuevo canvas
   912→    const img = await new Promise<HTMLImageElement | null>((resolve) => {
   913→      const i = new Image();
   914→      i.onload = () => resolve(i);
   915→      i.onerror = () => resolve(null);
   916→      i.src = drawing.dataUrl;
   917→    });
   918→    if (img) {
   919→      newCtx.drawImage(img, offsetX, offsetY);
   920→    }
   921→
   922→    const newDataUrl = newCanvas.toDataURL("image/png");
   923→    const now = Date.now();
   924→
   925→    // Pre-cachear la nueva imagen para que el rendering sea síncrono
   926→    try {
   927→      const { preloadImage } = await import("./useDrawingEngine");
   928→      await preloadImage(drawingId, newDataUrl);
   929→    } catch (_) {
   930→      // ignore
   931→    }
   932→
   933→    // Actualizar el drawing con las nuevas dimensiones y dataUrl
   934→    set((s) => {
   935→      if (!s.project) return {};
   936→      return {
   937→        project: {
   938→          ...s.project,
   939→          drawings: {
   940→            ...s.project.drawings,
   941→            [drawingId]: {
   942→              ...drawing,
   943→              dataUrl: newDataUrl,
   944→              width: newW,
   945→              height: newH,
   946→              updatedAt: now,
   947→            },
   948→          },
   949→          dirty: true,
   950→          updatedAt: now,
   951→        },
   952→      };
   953→    });
   954→
   955→    return { offsetX, offsetY };
   956→  },

  1101→  setCellDuration: (layerId, cellId, newDuration) => {
  1102→    // Cambia la duración (exposición) de un cell. Si la duración AUMENTA,
  1103→    // los cells posteriores de la misma capa se desplazan a la derecha
  1104→    // para no quedar solapados (que es el bug que reporta el usuario).
  1105→    // Si disminuye, los cells siguientes se "pegan" al nuevo fin
  1106→    // manteniendo la continuidad temporal (más útil que dejar un hueco).
  1107→    const state = get();
  1108→    if (!state.project) return;
  1109→    get().pushProjectSnapshot();
  1110→    set(updateProject(state, (p) => {
  1111→      const layers = p.layers.map((l) => {
  1112→        if (l.id !== layerId) return l;
  1113→        const idx = l.cells.findIndex((c) => c.id === cellId);
  1114→        if (idx < 0) return l;
  1115→        const cell = l.cells[idx];
  1116→        const safeDuration = Math.max(1, newDuration);
  1117→        const delta = safeDuration - cell.duration;
  1118→        // Mover todos los cells posteriores por delta
  1119→        const cells = l.cells.map((c, i) => {
  1120→          if (i === idx) return { ...c, duration: safeDuration };
  1121→          if (i > idx) return { ...c, startFrame: Math.max(0, c.startFrame + delta) };
  1122→          return c;
  1123→        });
  1124→        cells.sort((a, b) => a.startFrame - b.startFrame);
  1125→        return { ...l, cells };
  1126→      });
  1127→      return { ...p, layers };
  1128→    }));
  1129→  },

  1226→  moveCell: (layerId, cellId, newFrame) => {
  1227→    // Mueve un cell a `newFrame` (en coords absolutas del timeline). Si al
  1228→    // moverlo queda solapado con otro cell de la misma capa, empuja los
  1229→    // posteriores a la derecha hasta que no haya solapamiento. Esto
  1230→    // permite arrastrar cells dentro del timeline y que el layout se
  1231→    // reorganice sin superposiciones.
  1232→    const state = get();
  1233→    if (!state.project) return;
  1234→    get().pushProjectSnapshot();
  1235→    set(updateProject(state, (p) => {
  1236→      const layers = p.layers.map((l) => {
  1237→        if (l.id !== layerId) return l;
  1238→        const cells = l.cells.map((c) =>
  1239→          c.id === cellId ? { ...c, startFrame: Math.max(0, newFrame) } : c
  1240→        );
  1241→        cells.sort((a, b) => a.startFrame - b.startFrame);
  1242→        // Resolver solapamientos: si un cell empieza antes de que termine
  1243→        // el anterior, empujarlo a la derecha. Procesar en orden.
  1244→        for (let i = 1; i < cells.length; i++) {
  1245→          const prev = cells[i - 1];
  1246→          const cur = cells[i];
  1247→          const prevEnd = prev.startFrame + prev.duration;
  1248→          if (cur.startFrame < prevEnd) {
  1249→            cells[i] = { ...cur, startFrame: prevEnd };
  1250→          }
  1251→        }
  1252→        return { ...l, cells };
  1253→      });
  1254→      return { ...p, layers };
  1255→    }));
  1256→  },

  1653→  pushProjectSnapshot: () => {
  1654→    // Captura una referencia al project actual ANTES de mutarlo. Las
  1655→    // acciones estructurales (addLayer, moveCell, etc.) deben llamar esto
  1656→    // ANTES de hacer el `set(updateProject(...))`. Como el proyecto es
  1657→    // inmutable (cada mutación produce un nuevo objeto), la referencia
  1658→    // capturada sigue siendo el "estado previo" y se puede restaurar en
  1659→    // undo sin necesidad de diffing.
  1660→    set((s) => {
  1661→      if (!s.project) return {};
  1662→      return {
  1663→        undoStack: [...s.undoStack, { type: "project" as const, prevProject: s.project }].slice(-50),
  1664→        redoStack: [],
  1665→      };
  1666→    });
  1667→  },
  1668→
  1669→  undo: () => {
  1670→    const s = get();
  1671→    if (s.undoStack.length === 0) return;
  1672→    const entry = s.undoStack[s.undoStack.length - 1];
  1673→    if (entry.type === "drawing") {
  1674→      // Restaurar el dataUrl anterior en el proyecto
  1675→      set((st) => {
  1676→        if (!st.project) return {};
  1677→        if (!st.project.drawings[entry.drawingId]) return {};
  1678→        const restored = {
  1679→          ...st.project,
  1680→          drawings: {
  1681→            ...st.project.drawings,
  1682→            [entry.drawingId]: {
  1683→              ...st.project.drawings[entry.drawingId],
  1684→              dataUrl: entry.prevDataUrl,
  1685→              updatedAt: Date.now(),
  1686→            },
  1687→          },
  1688→          dirty: true,
  1689→          updatedAt: Date.now(),
  1690→        };
  1691→        return {
  1692→          project: restored,
  1693→          undoStack: st.undoStack.slice(0, -1),
  1694→          redoStack: [...st.redoStack, entry],
  1695→          skipReloadOnHistory: false,
  1696→        };
  1697→      });
  1698→      // Pre-cachear la imagen restaurada
  1699→      import("./useDrawingEngine").then(({ preloadImage }) => {
  1700→        preloadImage(entry.drawingId, entry.prevDataUrl);
  1701→      });
  1702→    } else {
  1703→      // entry.type === "project"
  1704→      // Restaurar el project capturado en el snapshot. Como el project es
  1705→      // inmutable, esta referencia sigue siendo el estado previo.
  1706→      // IMPORTANTE: guardar el estado ACTUAL en redoStack ANTES de
  1707→      // restaurar, así el redo puede volver adelante.
  1708→      set((st) => {
  1709→        const curProject = st.project;
  1710→        if (!curProject) return {};
  1711→        return {
  1712→          project: entry.prevProject,
  1713→          undoStack: st.undoStack.slice(0, -1),
  1714→          // Para redo del snapshot de project, guardamos la versión QUE
  1715→          // teníamos ANTES de restaurar (i.e., la post-mutación).
  1716→          redoStack: [...st.redoStack, { type: "project" as const, prevProject: curProject }],
  1717→          skipReloadOnHistory: false,
  1718→        };
  1719→      });
  1720→    }
  1721→  },
  1722→
  1723→  redo: () => {
  1724→    const s = get();
  1725→    if (s.redoStack.length === 0) return;
  1726→    const entry = s.redoStack[s.redoStack.length - 1];
  1727→    if (entry.type === "drawing") {
  1728→      set((st) => {
  1729→        if (!st.project) return {};
  1730→        if (!st.project.drawings[entry.drawingId]) return {};
  1731→        const restored = {
  1732→          ...st.project,
  1733→          drawings: {
  1734→            ...st.project.drawings,
  1735→            [entry.drawingId]: {
  1736→              ...st.project.drawings[entry.drawingId],
  1737→              dataUrl: entry.newDataUrl,
  1738→              updatedAt: Date.now(),
  1739→            },
  1740→          },
  1741→          dirty: true,
  1742→          updatedAt: Date.now(),
  1743→        };
  1744→        return {
  1745→          project: restored,
  1746→          redoStack: st.redoStack.slice(0, -1),
  1747→          undoStack: [...st.undoStack, entry],
  1748→          skipReloadOnHistory: false,
  1749→        };
  1750→      });
  1751→      import("./useDrawingEngine").then(({ preloadImage }) => {
  1752→        preloadImage(entry.drawingId, entry.newDataUrl);
  1753→      });
  1754→    } else {
  1755→      // entry.type === "project" — prevProject es la versión POST-mutación
  1756→      // (la que guardamos cuando hicimos undo). La restauramos y guardamos
  1757→      // el estado actual (pre-redo) en undoStack.
  1758→      set((st) => {
  1759→        const curProject = st.project;
  1760→        if (!curProject) return {};
  1761→        return {
  1762→          project: entry.prevProject,
  1763→          redoStack: st.redoStack.slice(0, -1),
  1764→          undoStack: [...st.undoStack, { type: "project" as const, prevProject: curProject }],
  1765→          skipReloadOnHistory: false,
  1766→        };
  1767→      });
  1768→    }
  1769→  },
  1770→
  1771→  canUndo: () => get().undoStack.length > 0,
  1772→  canRedo: () => get().redoStack.length > 0,
  1773→
  1774→  // -------------------------------------------------------------------------
  1775→  // Placement de imagen (modo transformación de imagen importada)
  1776→  // -------------------------------------------------------------------------
  1777→
  1778→  startImagePlacement: (dataUrl, nativeW, nativeH) => {
  1779→    const s = get();
  1780→    if (!s.project) return;
  1781→    // Calcular tamaño inicial preservando aspect ratio:
  1782→    // - Si la imagen es más chica que el canvas, usar tamaño nativo
  1783→    // - Si es más grande, escalar para que entre (con margen del 10%)
  1784→    const cw = s.project.settings.width;
  1785→    const ch = s.project.settings.height;
  1786→    const margin = 0.1; // 10% de margen
  1787→    const maxW = cw * (1 - 2 * margin);
  1788→    const maxH = ch * (1 - 2 * margin);
  1789→    let w = nativeW;
  1790→    let h = nativeH;
  1791→    if (w > maxW || h > maxH) {
  1792→      const scale = Math.min(maxW / w, maxH / h);
  1793→      w = Math.round(w * scale);
  1794→      h = Math.round(h * scale);
  1795→    }
  1796→    // Centrar en el canvas
  1797→    const x = Math.round((cw - w) / 2);
  1798→    const y = Math.round((ch - h) / 2);
  1799→    set({
  1800→      imagePlacement: {
  1801→        dataUrl,
  1802→        x,
  1803→        y,
  1804→        width: w,
  1805→        height: h,
  1806→        nativeW,
  1807→        nativeH,
  1808→      },
  1809→    });
  1810→  },
  1811→
  1812→  updateImagePlacement: (patch) => {
  1813→    set((s) => ({
  1814→      imagePlacement: s.imagePlacement ? { ...s.imagePlacement, ...patch } : null,
  1815→    }));
  1816→  },
  1817→
  1818→  confirmImagePlacement: async () => {
  1819→    const s = get();
  1820→    if (!s.project || !s.imagePlacement) return;
  1821→    const placement = s.imagePlacement;
  1822→    // Renderizar la imagen al tamaño actual en el canvas del drawing activo
  1823→    // Crear nuevo drawing o usar el existente
  1824→    const layer = s.project.layers.find((l) => l.id === s.project!.currentLayerId);
  1825→    if (!layer) {
  1826→      // No hay capa activa — no pegar, mostrar error claro
  1827→      set({ imagePlacement: null });
  1828→      console.warn("confirmImagePlacement: no hay capa activa");
  1829→      return;
  1830→    }
  1831→    if (layer.locked) {
  1832→      // Capa bloqueada — no se puede pegar, mostrar error claro
  1833→      set({ imagePlacement: null });
  1834→      alert(`La capa "${layer.name}" está bloqueada. Desbloqueala para pegar la imagen.`);
  1835→      return;
  1836→    }
  1837→    if (layer.type === "audio") {
  1838→      // Capa de audio — no acepta imágenes
  1839→      set({ imagePlacement: null });
  1840→      alert(`La capa "${layer.name}" es de audio. Seleccioná una capa de dibujo/referencia/fondo para pegar.`);
  1841→      return;
  1842→    }
  1843→    // Si la capa está OCULTA, la mostramos automáticamente — el usuario
  1844→    // está pegando explícitamente, así que quiere verla. Antes esto
  1845→    // abortaba silenciosamente y la imagen "desaparecía" sin explicación
  1846→    // (issue reportado: "no sé donde pega la imagen").
  1847→    if (!layer.visible) {
  1848→      set(updateProject(s, (p) => ({
  1849→        ...p,
  1850→        layers: p.layers.map((l) =>
  1851→          l.id === layer.id ? { ...l, visible: true } : l
  1852→        ),
  1853→      })));
  1854→    }
  1855→
  1856→    // Cargar la imagen
  1857→    const img = await new Promise<HTMLImageElement | null>((resolve) => {
  1858→      const i = new Image();
  1859→      i.onload = () => resolve(i);
  1860→      i.onerror = () => resolve(null);
  1861→      i.src = placement.dataUrl;
  1862→    });
  1863→    if (!img) {
  1864→      set({ imagePlacement: null });
  1865→      return;
  1866→    }
  1867→
  1868→    // Crear un canvas temporal con el contenido actual del drawing + la imagen pegada
  1869→    const tmpCanvas = document.createElement("canvas");
  1870→    tmpCanvas.width = s.project.settings.width;
  1871→    tmpCanvas.height = s.project.settings.height;
  1872→    const tmpCtx = tmpCanvas.getContext("2d")!;
  1873→    // Si hay un drawing existente, dibujarlo primero
  1874→    const cell = layer.cells.find(
  1875→      (c) =>
  1876→        s.project!.currentFrame >= c.startFrame &&
  1877→        s.project!.currentFrame < c.startFrame + c.duration
  1878→    );
  1879→    if (cell?.drawingId) {
  1880→      const existing = s.project.drawings[cell.drawingId];
  1881→      if (existing) {
  1882→        const existingImg = await new Promise<HTMLImageElement | null>((resolve) => {
  1883→          const i = new Image();
  1884→          i.onload = () => resolve(i);
  1885→          i.onerror = () => resolve(null);
  1886→          i.src = existing.dataUrl;
  1887→        });
  1888→        if (existingImg) {
  1889→          tmpCtx.drawImage(existingImg, 0, 0, tmpCanvas.width, tmpCanvas.height);
  1890→        }
  1891→      }
  1892→    }
  1893→    // Dibujar la imagen colocada en su transformación final
  1894→    tmpCtx.drawImage(img, placement.x, placement.y, placement.width, placement.height);
  1895→    const finalDataUrl = tmpCanvas.toDataURL("image/png");
  1896→
  1897→    // Crear/actualizar el drawing con la imagen pegada
  1898→    const drawingId = cell?.drawingId ?? crypto.randomUUID();
  1899→    const now = Date.now();
  1900→    const prevDataUrl = cell?.drawingId ? s.project.drawings[cell.drawingId]?.dataUrl ?? "" : "";
  1901→
  1902→    // Pre-cachear la imagen final
  1903→    const { preloadImage } = await import("./useDrawingEngine");
  1904→    await preloadImage(drawingId, finalDataUrl);
  1905→
  1906→    set((st) => {
  1907→      if (!st.project) return {};
  1908→      const layers = st.project.layers.map((l) => {
  1909→        if (l.id !== layer.id) return l;
  1910→        const cells = [...l.cells];
  1911→        const existingCell = cells.find(
  1912→          (c) =>
  1913→            st.project!.currentFrame >= c.startFrame &&
  1914→            st.project!.currentFrame < c.startFrame + c.duration
  1915→        );
  1916→        if (existingCell) {
  1917→          const idx = cells.indexOf(existingCell);
  1918→          cells[idx] = { ...existingCell, drawingId };
  1919→        } else {
  1920→          cells.push({
  1921→            id: crypto.randomUUID(),
  1922→            drawingId,
  1923→            startFrame: st.project.currentFrame,
  1924→            duration: 1,
  1925→          });
  1926→          cells.sort((a, b) => a.startFrame - b.startFrame);
  1927→        }
  1928→        return { ...l, cells };
  1929→      });
  1930→      return {
  1931→        project: {
  1932→          ...st.project,
  1933→          layers,
  1934→          drawings: {
  1935→            ...st.project.drawings,
  1936→            [drawingId]: {
  1937→              id: drawingId,
  1938→              name: "Imagen importada",
  1939→              dataUrl: finalDataUrl,
  1940→              width: st.project.settings.width,
  1941→              height: st.project.settings.height,
  1942→              createdAt: now,
  1943→              updatedAt: now,
  1944→            },
  1945→          },
  1946→          dirty: true,
  1947→          updatedAt: now,
  1948→        },
  1949→        imagePlacement: null,
  1950→      };
  1951→    });
  1952→
  1953→    // Registrar en el stack de undo
  1954→    if (prevDataUrl && prevDataUrl !== finalDataUrl) {
  1955→      get().pushHistory(drawingId, prevDataUrl, finalDataUrl);
  1956→    }
  1957→  },
  1958→
  1959→  cancelImagePlacement: () => {
  1960→    set({ imagePlacement: null });
  1961→  },
  1962→
  1963→  triggerAutosave: async () => {
  1964→    const s = get();
  1965→    if (!s.project || !s.project.dirty) return;
  1966→    set({ isAutosaving: true });
  1967→    await saveProjectToDb(s.project);
  1968→    set({
  1969→      isAutosaving: false,
  1970→      lastAutosaveAt: Date.now(),
  1971→      projects: [s.project, ...s.projects.filter((x) => x.id !== s.project!.id)],
  1972→    });
  1973→    set((st) => (st.project ? { project: { ...st.project, dirty: false } } : {}));
  1974→  },
  1975→}));
