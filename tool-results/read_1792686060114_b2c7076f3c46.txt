   261→// ---------------------------------------------------------------------------
   262→// Tipo para entradas del stack de undo/redo
   263→// ---------------------------------------------------------------------------
   264→// Dos tipos de entrada:
   265→//  - "drawing": edición in-place de un drawing (pincel, goma, etc.)
   266→//  - "project": cambio estructural (agregar capa, mover cell, extender
   267→//    cell, etc.). Para estos, capturamos una referencia al project
   268→//    PRE-mutación. Como los proyectos son inmutables (cada mutación
   269→//    produce un nuevo objeto), la referencia capturada sigue apuntando
   270→//    al estado anterior. La memoria es eficiente porque los drawings
   271→//    compartidos entre snapshots se referencian, no se duplican.
   272→
   273→type HistoryEntry =
   274→  | { type: "drawing"; drawingId: ID; prevDataUrl: string; newDataUrl: string }
   275→  | { type: "project"; prevProject: AnimationProject };
   276→
   277→// ---------------------------------------------------------------------------
   278→// Tipo para el placement de imagen (transformación interactiva)
   279→// ---------------------------------------------------------------------------
   280→
   281→interface ImagePlacement {
   282→  dataUrl: string;        // imagen original (data URL)
   283→  x: number;               // posición x en coords del canvas (esquina sup-izq)
   284→  y: number;               // posición y
   285→  width: number;           // ancho actual (preserva aspect ratio)
   286→  height: number;          // alto actual
   287→  nativeW: number;         // ancho original de la imagen (px)
   288→  nativeH: number;         // alto original (px)
   289→}

   294→
   295→function updateProject(
   296→  state: AppState,
   297→  updater: (p: AnimationProject) => AnimationProject
   298→): Partial<AppState> {
   299→  if (!state.project) return {};
   300→  const newProject = updater({ ...state.project, dirty: true, updatedAt: Date.now() });
   301→  return { project: newProject };
   302→}

   766→          kfs.push(newKf);
   767→          kfs.sort((a, b) => a.frame - b.frame);
   768→        }

   770→      }),

   801→      const endFrame = Math.max(anchor.startFrame, clickedCell.startFrame);
   802→      const range = layer.cells.filter(
   803→        (c) => c.startFrame >= startFrame && c.startFrame <= endFrame
   804→      );

   809→      // Ctrl/Cmd+click: toggle individual
   810→      const newSet = new Set(state.selectedCellIds);
   811→      if (newSet.has(cellId)) newSet.delete(cellId);
   812→      else newSet.add(cellId);
   813→      set({ selectedCellIds: newSet });
   814→    } else {
   815→      // Click normal: selección única
   816→      set({ selectedCellIds: new Set([cellId]) });

   952→    // los cells posteriores de la misma capa se desplazan a la derecha

  1075→  moveCell: (layerId, cellId, newFrame) => {
  1076→    // Mueve un cell a `newFrame` (en coords absolutas del timeline). Si al
  1077→    // moverlo queda solapado con otro cell de la misma capa, empuja los
  1078→    // posteriores a la derecha hasta que no haya solapamiento. Esto
  1079→    // permite arrastrar cells dentro del timeline y que el layout se
  1080→    // reorganice sin superposiciones.
  1081→    const state = get();
  1082→    if (!state.project) return;
  1083→    get().pushProjectSnapshot();
  1084→    set(updateProject(state, (p) => {
  1085→      const layers = p.layers.map((l) => {
  1086→        if (l.id !== layerId) return l;
  1087→        const cells = l.cells.map((c) =>
  1088→          c.id === cellId ? { ...c, startFrame: Math.max(0, newFrame) } : c
  1089→        );
  1090→        cells.sort((a, b) => a.startFrame - b.startFrame);
  1091→        // Resolver solapamientos: si un cell empieza antes de que termine
  1092→        // el anterior, empujarlo a la derecha. Procesar en orden.

  1099→          }
  1100→        }

  1102→      });
  1103→      return { ...p, layers };
  1104→    }));

  1522→    if (entry.type === "drawing") {

  1554→      // inmutable, esta referencia sigue siendo el estado previo.

  1627→  startImagePlacement: (dataUrl, nativeW, nativeH) => {
  1628→    const s = get();
  1629→    if (!s.project) return;
  1630→    // Calcular tamaño inicial preservando aspect ratio:
  1631→    // - Si la imagen es más chica que el canvas, usar tamaño nativo
  1632→    // - Si es más grande, escalar para que entre (con margen del 10%)
  1633→    const cw = s.project.settings.width;
  1634→    const ch = s.project.settings.height;
  1635→    const margin = 0.1; // 10% de margen
  1636→    const maxW = cw * (1 - 2 * margin);
  1637→    const maxH = ch * (1 - 2 * margin);
  1638→    let w = nativeW;
  1639→    let h = nativeH;
  1640→    if (w > maxW || h > maxH) {
  1641→      const scale = Math.min(maxW / w, maxH / h);
  1642→      w = Math.round(w * scale);
  1643→      h = Math.round(h * scale);
  1644→    }
  1645→    // Centrar en el canvas
  1646→    const x = Math.round((cw - w) / 2);
  1647→    const y = Math.round((ch - h) / 2);
  1648→    set({
  1649→      imagePlacement: {
  1650→        dataUrl,
  1651→        x,
  1652→        y,
  1653→        width: w,
  1654→        height: h,
  1655→        nativeW,
  1656→        nativeH,
  1657→      },
  1658→    });

  1667→  confirmImagePlacement: async () => {

  1669→    if (!s.project || !s.imagePlacement) return;
  1670→    const placement = s.imagePlacement;
  1671→    // Renderizar la imagen al tamaño actual en el canvas del drawing activo
  1672→    // Crear nuevo drawing o usar el existente
  1673→    const layer = s.project.layers.find((l) => l.id === s.project!.currentLayerId);
  1674→    if (!layer || layer.locked || !layer.visible || layer.type === "audio") {
  1675→      set({ imagePlacement: null });
  1676→      return;
  1677→    }
  1678→
  1679→    // Cargar la imagen
  1680→    const img = await new Promise<HTMLImageElement | null>((resolve) => {
  1681→      const i = new Image();
  1682→      i.onload = () => resolve(i);
  1683→      i.onerror = () => resolve(null);
  1684→      i.src = placement.dataUrl;
  1685→    });
  1686→    if (!img) {
  1687→      set({ imagePlacement: null });
  1688→      return;
  1689→    }
  1690→
  1691→    // Crear un canvas temporal con el contenido actual del drawing + la imagen pegada
  1692→    const tmpCanvas = document.createElement("canvas");
  1693→    tmpCanvas.width = s.project.settings.width;
  1694→    tmpCanvas.height = s.project.settings.height;
  1695→    const tmpCtx = tmpCanvas.getContext("2d")!;
  1696→    // Si hay un drawing existente, dibujarlo primero
  1697→    const cell = layer.cells.find(
  1698→      (c) =>
  1699→        s.project!.currentFrame >= c.startFrame &&
  1700→        s.project!.currentFrame < c.startFrame + c.duration
  1701→    );
  1702→    if (cell?.drawingId) {
  1703→      const existing = s.project.drawings[cell.drawingId];
  1704→      if (existing) {
  1705→        const existingImg = await new Promise<HTMLImageElement | null>((resolve) => {
  1706→          const i = new Image();
  1707→          i.onload = () => resolve(i);
  1708→          i.onerror = () => resolve(null);
  1709→          i.src = existing.dataUrl;
  1710→        });
  1711→        if (existingImg) {
  1712→          tmpCtx.drawImage(existingImg, 0, 0, tmpCanvas.width, tmpCanvas.height);
  1713→        }
  1714→      }
  1715→    }
  1716→    // Dibujar la imagen colocada en su transformación final
  1717→    tmpCtx.drawImage(img, placement.x, placement.y, placement.width, placement.height);
  1718→    const finalDataUrl = tmpCanvas.toDataURL("image/png");
  1719→
  1720→    // Crear/actualizar el drawing con la imagen pegada
  1721→    const drawingId = cell?.drawingId ?? crypto.randomUUID();
  1722→    const now = Date.now();
  1723→    const prevDataUrl = cell?.drawingId ? s.project.drawings[cell.drawingId]?.dataUrl ?? "" : "";
  1724→
  1725→    // Pre-cachear la imagen final
  1726→    const { preloadImage } = await import("./useDrawingEngine");
  1727→    await preloadImage(drawingId, finalDataUrl);
  1728→
  1729→    set((st) => {
  1730→      if (!st.project) return {};
  1731→      const layers = st.project.layers.map((l) => {
  1732→        if (l.id !== layer.id) return l;
  1733→        const cells = [...l.cells];
  1734→        const existingCell = cells.find(
  1735→          (c) =>
  1736→            st.project!.currentFrame >= c.startFrame &&
  1737→            st.project!.currentFrame < c.startFrame + c.duration
  1738→        );
  1739→        if (existingCell) {
  1740→          const idx = cells.indexOf(existingCell);
  1741→          cells[idx] = { ...existingCell, drawingId };
  1742→        } else {
  1743→          cells.push({
  1744→            id: crypto.randomUUID(),
  1745→            drawingId,
  1746→            startFrame: st.project.currentFrame,
  1747→            duration: 1,
  1748→          });
  1749→          cells.sort((a, b) => a.startFrame - b.startFrame);
  1750→        }
  1751→        return { ...l, cells };
  1752→      });
  1753→      return {
  1754→        project: {
  1755→          ...st.project,
  1756→          layers,
  1757→          drawings: {
  1758→            ...st.project.drawings,
  1759→            [drawingId]: {
  1760→              id: drawingId,
  1761→              name: "Imagen importada",
  1762→              dataUrl: finalDataUrl,
  1763→              width: st.project.settings.width,
  1764→              height: st.project.settings.height,
  1765→              createdAt: now,
  1766→              updatedAt: now,
  1767→            },
  1768→          },
  1769→          dirty: true,
  1770→          updatedAt: now,
  1771→        },
  1772→        imagePlacement: null,
  1773→      };
  1774→    });
  1775→
  1776→    // Registrar en el stack de undo
  1777→    if (prevDataUrl && prevDataUrl !== finalDataUrl) {
  1778→      get().pushHistory(drawingId, prevDataUrl, finalDataUrl);
  1779→    }
  1780→  },
  1781→
  1782→  cancelImagePlacement: () => {
  1783→    set({ imagePlacement: null });
  1784→  },
  1785→
  1786→  triggerAutosave: async () => {
  1787→    const s = get();
  1788→    if (!s.project || !s.project.dirty) return;
  1789→    set({ isAutosaving: true });
  1790→    await saveProjectToDb(s.project);
  1791→    set({
  1792→      isAutosaving: false,
  1793→      lastAutosaveAt: Date.now(),
  1794→      projects: [s.project, ...s.projects.filter((x) => x.id !== s.project!.id)],
  1795→    });
  1796→    set((st) => (st.project ? { project: { ...st.project, dirty: false } } : {}));
  1797→  },
  1798→}));
