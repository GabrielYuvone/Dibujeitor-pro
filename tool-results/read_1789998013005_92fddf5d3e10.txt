   223→interface ImagePlacement {
   224→  dataUrl: string;        // imagen original (data URL)
   225→  x: number;               // posición x en coords del canvas (esquina sup-izq)
   226→  y: number;               // posición y
   227→  width: number;           // ancho actual (preserva aspect ratio)
   228→  height: number;          // alto actual
   229→  nativeW: number;         // ancho original de la imagen (px)
   230→  nativeH: number;         // alto original (px)

   233→// ---------------------------------------------------------------------------
   234→// Helpers internos
   235→// ---------------------------------------------------------------------------
   236→
   237→function updateProject(
   238→  state: AppState,
   239→  updater: (p: AnimationProject) => AnimationProject
   240→): Partial<AppState> {
   241→  if (!state.project) return {};
   242→  const newProject = updater({ ...state.project, dirty: true, updatedAt: Date.now() });
   243→  return { project: newProject };
   244→}

   494→  moveLayer: (id, dir) => {
   495→    const state = get();
   496→    if (!state.project) return;
   497→    set(updateProject(state, (p) => {
   498→      const layers = [...p.layers];
   499→      const idx = layers.findIndex((l) => l.id === id);
   500→      if (idx < 0) return p;
   501→      const target = dir === "up" ? idx - 1 : idx + 1;
   502→      if (target < 0 || target >= layers.length) return p;
   503→      [layers[idx], layers[target]] = [layers[target], layers[idx]];
   504→      return { ...p, layers };
   505→    }));
   506→  },
   507→
   508→  duplicateLayer: (id) => {
   509→    const state = get();
   510→    if (!state.project) return;
   511→    set(updateProject(state, (p) => {
   512→      const layer = p.layers.find((l) => l.id === id);
   513→      if (!layer) return p;
   514→      const newId = genId("layer");
   515→      const newLayer: Layer = {
   516→        ...layer,
   517→        id: newId,
   518→        name: `${layer.name} (copia)`,
   519→        cells: layer.cells.map((c) => ({ ...c, id: genId("cell") })),
   520→        createdAt: Date.now(),
   521→      };
   522→      const idx = p.layers.findIndex((l) => l.id === id);
   523→      const layers = [...p.layers];
   524→      layers.splice(idx + 1, 0, newLayer);
   525→      return { ...p, layers, currentLayerId: newId };
   526→    }));
   527→  },

   529→  // -------------------------------------------------------------------------
   530→  // Cells / drawings
   531→  // -------------------------------------------------------------------------
   532→
   533→  ensureDrawingForCell: async (layerId, frame) => {
   534→    const state = get();
   535→    if (!state.project) throw new Error("No project");
   536→    const layer = state.project.layers.find((l) => l.id === layerId);
   537→    if (!layer) throw new Error("Layer not found");
   538→
   539→    // Buscar celda existente en este frame
   540→    const existing = layer.cells.find(
   541→      (c) => frame >= c.startFrame && frame < c.startFrame + c.duration
   542→    );
   543→
   544→    if (existing && existing.drawingId) {
   545→      return existing.drawingId;
   546→    }
   547→
   548→    // Crear nuevo Drawing vacío
   549→    const drawingId = genId("draw");
   550→    const now = Date.now();
   551→    const canvas = document.createElement("canvas");
   552→    canvas.width = state.project.settings.width;
   553→    canvas.height = state.project.settings.height;
   554→    const dataUrl = canvas.toDataURL("image/png");
   555→
   556→    const drawing: Drawing = {
   557→      id: drawingId,
   558→      name: `Drawing ${Object.keys(state.project.drawings).length + 1}`,
   559→      dataUrl,
   560→      width: state.project.settings.width,
   561→      height: state.project.settings.height,
   562→      createdAt: now,
   563→      updatedAt: now,
   564→    };
   565→
   566→    set(updateProject(state, (p) => {
   567→      const layers = p.layers.map((l) => {
   568→        if (l.id !== layerId) return l;
   569→        const cells = [...l.cells];
   570→        // Si existe celda en este frame, asignarle el drawingId
   571→        const cell = cells.find(
   572→          (c) => frame >= c.startFrame && frame < c.startFrame + c.duration
   573→        );
   574→        if (cell) {
   575→          const idx = cells.indexOf(cell);
   576→          cells[idx] = { ...cell, drawingId };
   577→        } else {
   578→          // Crear nueva celda
   579→          cells.push({ id: genId("cell"), drawingId, startFrame: frame, duration: 1 });
   580→          cells.sort((a, b) => a.startFrame - b.startFrame);
   581→        }
   582→        return { ...l, cells };
   583→      });
   584→      return {
   585→        ...p,
   586→        layers,
   587→        drawings: { ...p.drawings, [drawingId]: drawing },
   588→      };
   589→    }));
   590→
   591→    return drawingId;
   592→  },
   593→

  1174→  startImagePlacement: (dataUrl, nativeW, nativeH) => {
  1175→    const s = get();
  1176→    if (!s.project) return;
  1177→    // Calcular tamaño inicial preservando aspect ratio:
  1178→    // - Si la imagen es más chica que el canvas, usar tamaño nativo
  1179→    // - Si es más grande, escalar para que entre (con margen del 10%)
  1180→    const cw = s.project.settings.width;
  1181→    const ch = s.project.settings.height;
  1182→    const margin = 0.1; // 10% de margen
  1183→    const maxW = cw * (1 - 2 * margin);
  1184→    const maxH = ch * (1 - 2 * margin);
  1185→    let w = nativeW;
  1186→    let h = nativeH;
  1187→    if (w > maxW || h > maxH) {
  1188→      const scale = Math.min(maxW / w, maxH / h);
  1189→      w = Math.round(w * scale);
  1190→      h = Math.round(h * scale);
  1191→    }
  1192→    // Centrar en el canvas
  1193→    const x = Math.round((cw - w) / 2);
  1194→    const y = Math.round((ch - h) / 2);
  1195→    set({
  1196→      imagePlacement: {

  1199→        y,
  1200→        width: w,

  1208→  updateImagePlacement: (patch) => {

  1214→  confirmImagePlacement: async () => {
  1215→    const s = get();
  1216→    if (!s.project || !s.imagePlacement) return;
  1217→    const placement = s.imagePlacement;
  1218→    // Renderizar la imagen al tamaño actual en el canvas del drawing activo
  1219→    // Crear nuevo drawing o usar el existente
  1220→    const layer = s.project.layers.find((l) => l.id === s.project!.currentLayerId);
  1221→    if (!layer || layer.locked || !layer.visible || layer.type === "audio") {
  1222→      set({ imagePlacement: null });
  1223→      return;
  1224→    }
  1225→
  1226→    // Cargar la imagen
  1227→    const img = await new Promise<HTMLImageElement | null>((resolve) => {
  1228→      const i = new Image();
  1229→      i.onload = () => resolve(i);
  1230→      i.onerror = () => resolve(null);
  1231→      i.src = placement.dataUrl;
  1232→    });
  1233→    if (!img) {
  1234→      set({ imagePlacement: null });
  1235→      return;
  1236→    }
  1237→
  1238→    // Crear un canvas temporal con el contenido actual del drawing + la imagen pegada
  1239→    const tmpCanvas = document.createElement("canvas");
  1240→    tmpCanvas.width = s.project.settings.width;
  1241→    tmpCanvas.height = s.project.settings.height;
  1242→    const tmpCtx = tmpCanvas.getContext("2d")!;
  1243→    // Si hay un drawing existente, dibujarlo primero
  1244→    const cell = layer.cells.find(
  1245→      (c) =>
  1246→        s.project!.currentFrame >= c.startFrame &&
  1247→        s.project!.currentFrame < c.startFrame + c.duration
  1248→    );
  1249→    if (cell?.drawingId) {
  1250→      const existing = s.project.drawings[cell.drawingId];
  1251→      if (existing) {
  1252→        const existingImg = await new Promise<HTMLImageElement | null>((resolve) => {
  1253→          const i = new Image();
  1254→          i.onload = () => resolve(i);
  1255→          i.onerror = () => resolve(null);
  1256→          i.src = existing.dataUrl;
  1257→        });
  1258→        if (existingImg) {
  1259→          tmpCtx.drawImage(existingImg, 0, 0, tmpCanvas.width, tmpCanvas.height);
  1260→        }
  1261→      }
  1262→    }
  1263→    // Dibujar la imagen colocada en su transformación final
  1264→    tmpCtx.drawImage(img, placement.x, placement.y, placement.width, placement.height);
  1265→    const finalDataUrl = tmpCanvas.toDataURL("image/png");
  1266→
  1267→    // Crear/actualizar el drawing con la imagen pegada
  1268→    const drawingId = cell?.drawingId ?? crypto.randomUUID();
  1269→    const now = Date.now();
  1270→    const prevDataUrl = cell?.drawingId ? s.project.drawings[cell.drawingId]?.dataUrl ?? "" : "";
  1271→
  1272→    // Pre-cachear la imagen final
  1273→    const { preloadImage } = await import("./useDrawingEngine");
  1274→    await preloadImage(drawingId, finalDataUrl);
  1275→
  1276→    set((st) => {
  1277→      if (!st.project) return {};
  1278→      const layers = st.project.layers.map((l) => {
  1279→        if (l.id !== layer.id) return l;
  1280→        const cells = [...l.cells];
  1281→        const existingCell = cells.find(
  1282→          (c) =>
  1283→            st.project!.currentFrame >= c.startFrame &&
  1284→            st.project!.currentFrame < c.startFrame + c.duration
  1285→        );
  1286→        if (existingCell) {
  1287→          const idx = cells.indexOf(existingCell);
  1288→          cells[idx] = { ...existingCell, drawingId };
  1289→        } else {
  1290→          cells.push({
  1291→            id: crypto.randomUUID(),
  1292→            drawingId,
  1293→            startFrame: st.project.currentFrame,
  1294→            duration: 1,
  1295→          });
  1296→          cells.sort((a, b) => a.startFrame - b.startFrame);
  1297→        }
  1298→        return { ...l, cells };
  1299→      });
  1300→      return {
  1301→        project: {
  1302→          ...st.project,
  1303→          layers,
  1304→          drawings: {
  1305→            ...st.project.drawings,
  1306→            [drawingId]: {
  1307→              id: drawingId,
  1308→              name: "Imagen importada",
  1309→              dataUrl: finalDataUrl,
  1310→              width: st.project.settings.width,
  1311→              height: st.project.settings.height,
  1312→              createdAt: now,
  1313→              updatedAt: now,
  1314→            },
  1315→          },
  1316→          dirty: true,
  1317→          updatedAt: now,
  1318→        },
  1319→        imagePlacement: null,
  1320→      };
  1321→    });
  1322→
  1323→    // Registrar en el stack de undo
  1324→    if (prevDataUrl && prevDataUrl !== finalDataUrl) {
  1325→      get().pushHistory(drawingId, prevDataUrl, finalDataUrl);
  1326→    }
  1327→  },
  1328→
  1329→  cancelImagePlacement: () => {
  1330→    set({ imagePlacement: null });
  1331→  },
  1332→
  1333→  triggerAutosave: async () => {

  1336→    set({ isAutosaving: true });

  1342→    });
  1343→    set((st) => (st.project ? { project: { ...st.project, dirty: false } } : {}));
  1344→  },
