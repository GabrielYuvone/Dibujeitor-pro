   245→// ---------------------------------------------------------------------------
   246→// Helpers internos
   247→// ---------------------------------------------------------------------------
   248→
   249→function updateProject(
   250→  state: AppState,
   251→  updater: (p: AnimationProject) => AnimationProject
   252→): Partial<AppState> {
   253→  if (!state.project) return {};
   254→  const newProject = updater({ ...state.project, dirty: true, updatedAt: Date.now() });
   255→  return { project: newProject };
   256→}

   278→  canvasView: { zoom: 1, panX: 0, panY: 0, rotation: 0 },

   280→  exportOptions: {
   281→    format: "mp4",
   282→    width: DEFAULT_SETTINGS.width,
   283→    height: DEFAULT_SETTINGS.height,
   284→    fps: DEFAULT_SETTINGS.fps,
   285→    rangeStart: 0,
   286→    rangeEnd: 0,
   287→    includeAudio: true,
   288→    quality: 0.92,
   289→    backgroundColor: "#ffffff",
   290→  },
   291→
   292→  panelLayout: "default",
   293→  showGrid: false,
   294→  showSafeArea: false,
   295→  showRulers: false,
   296→  thumbnailCache: {},
   297→  isAutosaving: false,
   298→  lastAutosaveAt: null,
   299→  drawingBuffer: null,
   300→  // Estado de dibujo y modificadores de teclado (no persistidos)
   301→  isDrawing: false,
   302→  rightPanelVisible: true,
   303→  xModifier: false,
   304→  zModifier: false,
   305→  toolSettings: {} as Record<string, Partial<BrushSettings>>,
   306→  customPalette: [] as string[],
   307→  // Pilas de undo/redo (no persistidas)
   308→  undoStack: [] as HistoryEntry[],
   309→  redoStack: [] as HistoryEntry[],
   310→  // Indicador para evitar que el motor de dibujo recargue el canvas
   311→  // cuando se hace undo/redo (la imagen cacheada ya está actualizada)
   312→  skipReloadOnHistory: false,
   313→  // Placement de imagen (transformación interactiva)
   314→  imagePlacement: null as ImagePlacement | null,
   315→
   316→  // -------------------------------------------------------------------------
   317→  // Projects
   318→  // -------------------------------------------------------------------------
   319→
   320→  refreshProjects: async () => {
   321→    set({ projectsLoading: true });
   322→    const projects = await listProjectsFromDb();
   323→    set({ projects, projectsLoading: false });
   324→  },
   325→
   326→  newProject: async (name) => {
   327→    const project = createBlankProject(name);
   328→    await saveProjectToDb(project);
   329→    set({ project, projects: [project, ...get().projects] });
   330→    return project;
   331→  },
   332→
   333→  openProject: async (id) => {
   334→    const project = await loadProjectFromDb(id);
   335→    if (project) {
   336→      set({ project, canvasView: { zoom: 1, panX: 0, panY: 0, rotation: 0 } });
   337→    }
   338→  },
   339→
   340→  saveCurrent: async () => {
   341→    const p = get().project;
   342→    if (!p) return;
   343→    await saveProjectToDb(p);
   344→    set({
   345→      projects: [p, ...get().projects.filter((x) => x.id !== p.id)],
   346→      isAutosaving: false,
   347→      lastAutosaveAt: Date.now(),
   348→    });
   349→    set((s) => (s.project ? { project: { ...s.project, dirty: false } } : {}));
   350→  },
   351→
   352→  duplicateCurrent: async (newName) => {
   353→    const p = get().project;
   354→    if (!p) return;
   355→    const copy = await duplicateProjectInDb(p.id, newName);
   356→    if (copy) {
   357→      set({ project: copy, projects: [copy, ...get().projects] });
   358→    }
   359→  },
   360→
   361→  deleteProject: async (id) => {
   362→    await deleteProjectFromDb(id);
   363→    const p = get().project;
   364→    if (p && p.id === id) {
   365→      set({ project: null });
   366→    }
   367→    set({ projects: get().projects.filter((x) => x.id !== id) });
   368→  },
   369→
   370→  closeProject: () => set({ project: null }),
   371→
   372→  exportProject: () => get().project,
   373→
   374→  // -------------------------------------------------------------------------
   375→  // Tools
   376→  // -------------------------------------------------------------------------
   377→
   378→  setTool: (tool) => {
   379→    set((s) => {
   380→      // Guardar la configuración actual del pincel para la herramienta actual
   381→      // y restaurar la configuración guardada para la nueva herramienta.
   382→      const prevTool = s.currentTool;
   383→      const toolSettings = { ...s.toolSettings };
   384→      // Guardar settings relevantes (size, opacity, color, hardness, etc.)
   385→      toolSettings[prevTool] = {
   386→        size: s.brush.size,
   387→        color: s.brush.color,
   388→        opacity: s.brush.opacity,
   389→        hardness: s.brush.hardness,
   390→        smoothing: s.brush.smoothing,
   391→        fillTolerance: s.brush.fillTolerance,
   392→      };
   393→      // Restaurar settings de la nueva herramienta (si existen)
   394→      const restored = toolSettings[tool];
   395→      const newBrush = restored
   396→        ? { ...s.brush, ...restored }
   397→        : s.brush;
   398→      return { currentTool: tool, toolSettings, brush: newBrush };
   399→    });
   400→  },
   401→  setBrush: (b) => set((s) => ({ brush: { ...s.brush, ...b } })),
   402→
   403→  setOnion: (o) => set((s) => ({ onion: { ...s.onion, ...o } })),
   404→
   405→  togglePlay: () => set((s) => ({ playback: { ...s.playback, playing: !s.playback.playing } })),
   406→  setPlaying: (p) => set((s) => ({ playback: { ...s.playback, playing: p } })),
   407→  setLooping: (l) => set((s) => ({ playback: { ...s.playback, looping: l } })),
   408→  setPlaybackFps: (fps) => set((s) => ({ playback: { ...s.playback, playbackFps: Math.max(1, fps) } })),
   409→  setRange: (start, end) =>
   410→    set((s) => ({ playback: { ...s.playback, rangeStart: start, rangeEnd: end } })),
   411→
   412→  setCanvasView: (v) => set((s) => ({ canvasView: { ...s.canvasView, ...v } })),
   413→  resetCanvasView: () => set({ canvasView: { zoom: 1, panX: 0, panY: 0, rotation: 0 } }),
   414→
   415→  gotoFrame: (frame) =>
   416→    set((s) =>
   417→      s.project
   418→        ? { project: { ...s.project, currentFrame: Math.max(0, frame) } }
   419→        : {}
   420→    ),
   421→  nextFrame: () =>
   422→    set((s) =>
   423→      s.project
   424→        ? { project: { ...s.project, currentFrame: s.project.currentFrame + 1 } }
   425→        : {}
   426→    ),
   427→  prevFrame: () =>
   428→    set((s) =>
   429→      s.project && s.project.currentFrame > 0
   430→        ? { project: { ...s.project, currentFrame: s.project.currentFrame - 1 } }
   431→        : {}
   432→    ),
   433→  setViewMode: (m) => set({ viewMode: m }),
   434→
   435→  setProjectSettings: (settings) =>
   436→    set((s) =>
   437→      s.project
   438→        ? {
   439→            project: {
   440→              ...s.project,
   441→              settings: { ...s.project.settings, ...settings },
   442→              dirty: true,
   443→              updatedAt: Date.now(),
   444→            },
   445→          }
   446→        : {}
   447→    ),
   448→
   449→  renameProject: (name) =>
   450→    set((s) =>
   451→      s.project
   452→        ? { project: { ...s.project, name, dirty: true, updatedAt: Date.now() } }
   453→        : {}
   454→    ),
   455→
   456→  // -------------------------------------------------------------------------
   457→  // Layers
   458→  // -------------------------------------------------------------------------
   459→
   460→  addLayer: (type, name) => {
   461→    const state = get();
   462→    if (!state.project) return "";
   463→    const id = genId("layer");
   464→    const newLayer: Layer = {
   465→      id,
   466→      name: name ?? `${type === "draw" ? "Capa de dibujo" : type === "reference" ? "Referencia" : type === "background" ? "Fondo" : "Audio"} ${state.project.layers.length + 1}`,
   467→      type,
   468→      cells: type === "draw" || type === "reference" || type === "background"
   469→        ? [{ id: genId("cell"), drawingId: null, startFrame: 0, duration: 1 }]
   470→        : [],
   471→      visible: true,
   472→      locked: false,
   473→      opacity: 1,
   474→      createdAt: Date.now(),
   475→    };
   476→    set(updateProject(state, (p) => ({
   477→      ...p,
   478→      layers: [...p.layers, newLayer],
   479→      currentLayerId: id,
   480→    })));
   481→    return id;
   482→  },
   483→
   484→  deleteLayer: (id) => {
   485→    const state = get();
   486→    if (!state.project) return;
   487→    if (state.project.layers.length <= 1) return;
   488→    set(updateProject(state, (p) => ({
   489→      ...p,
   490→      layers: p.layers.filter((l) => l.id !== id),
   491→      currentLayerId: p.currentLayerId === id ? p.layers[0].id : p.currentLayerId,
   492→    })));
   493→  },
   494→
   495→  selectLayer: (id) =>
   496→    set((s) =>
   497→      s.project ? { project: { ...s.project, currentLayerId: id } } : {}
   498→    ),
   499→
   500→  renameLayer: (id, name) => {
   501→    const state = get();
   502→    if (!state.project) return;
   503→    set(updateProject(state, (p) => ({
   504→      ...p,
   505→      layers: p.layers.map((l) => (l.id === id ? { ...l, name } : l)),
   506→    })));
   507→  },
   508→
   509→  toggleLayerVisible: (id) => {
   510→    const state = get();
   511→    if (!state.project) return;
   512→    set(updateProject(state, (p) => ({
   513→      ...p,
   514→      layers: p.layers.map((l) => (l.id === id ? { ...l, visible: !l.visible } : l)),
   515→    })));
   516→  },
   517→
   518→  toggleLayerLocked: (id) => {
   519→    const state = get();
   520→    if (!state.project) return;
   521→    set(updateProject(state, (p) => ({
   522→      ...p,
   523→      layers: p.layers.map((l) => (l.id === id ? { ...l, locked: !l.locked } : l)),
   524→    })));
   525→  },
   526→
   527→  setLayerOpacity: (id, opacity) => {
   528→    const state = get();
   529→    if (!state.project) return;
   530→    set(updateProject(state, (p) => ({
   531→      ...p,
   532→      layers: p.layers.map((l) => (l.id === id ? { ...l, opacity } : l)),
   533→    })));
   534→  },
   535→
   536→  moveLayer: (id, dir) => {
   537→    const state = get();
   538→    if (!state.project) return;
   539→    set(updateProject(state, (p) => {
   540→      const layers = [...p.layers];
   541→      const idx = layers.findIndex((l) => l.id === id);
   542→      if (idx < 0) return p;
   543→      const target = dir === "up" ? idx - 1 : idx + 1;
   544→      if (target < 0 || target >= layers.length) return p;
   545→      [layers[idx], layers[target]] = [layers[target], layers[idx]];
   546→      return { ...p, layers };
   547→    }));
   548→  },
   549→
   550→  duplicateLayer: (id) => {
   551→    const state = get();
   552→    if (!state.project) return;
   553→    set(updateProject(state, (p) => {
   554→      const layer = p.layers.find((l) => l.id === id);
   555→      if (!layer) return p;
   556→      const newId = genId("layer");
   557→      const newLayer: Layer = {
   558→        ...layer,
   559→        id: newId,
   560→        name: `${layer.name} (copia)`,
   561→        cells: layer.cells.map((c) => ({ ...c, id: genId("cell") })),
   562→        createdAt: Date.now(),
   563→      };
   564→      const idx = p.layers.findIndex((l) => l.id === id);
   565→      const layers = [...p.layers];
   566→      layers.splice(idx + 1, 0, newLayer);
   567→      return { ...p, layers, currentLayerId: newId };
   568→    }));
   569→  },
   570→
   571→  setLayerLoop: (id, range) => {
   572→    const state = get();
   573→    if (!state.project) return;
   574→    set(updateProject(state, (p) => ({
   575→      ...p,
   576→      layers: p.layers.map((l) => (l.id === id ? { ...l, loopRange: range } : l)),
   577→    })));
   578→  },
   579→
   580→  // -------------------------------------------------------------------------
   581→  // Cells / drawings
   582→  // -------------------------------------------------------------------------
   583→
   584→  ensureDrawingForCell: async (layerId, frame) => {
   585→    const state = get();
   586→    if (!state.project) throw new Error("No project");
   587→    const layer = state.project.layers.find((l) => l.id === layerId);
   588→    if (!layer) throw new Error("Layer not found");
   589→
   590→    // Buscar celda existente en este frame
   591→    const existing = layer.cells.find(
   592→      (c) => frame >= c.startFrame && frame < c.startFrame + c.duration
   593→    );
   594→
   595→    if (existing && existing.drawingId) {
   596→      return existing.drawingId;
   597→    }
   598→
   599→    // Crear nuevo Drawing vacío
   600→    const drawingId = genId("draw");
   601→    const now = Date.now();
   602→    const canvas = document.createElement("canvas");
   603→    canvas.width = state.project.settings.width;
   604→    canvas.height = state.project.settings.height;
   605→    const dataUrl = canvas.toDataURL("image/png");
   606→
   607→    const drawing: Drawing = {
   608→      id: drawingId,
   609→      name: `Drawing ${Object.keys(state.project.drawings).length + 1}`,
   610→      dataUrl,
   611→      width: state.project.settings.width,
   612→      height: state.project.settings.height,
   613→      createdAt: now,
   614→      updatedAt: now,
   615→    };
   616→
   617→    set(updateProject(state, (p) => {
   618→      const layers = p.layers.map((l) => {
   619→        if (l.id !== layerId) return l;
   620→        const cells = [...l.cells];
   621→        // Si existe celda en este frame, asignarle el drawingId
   622→        const cell = cells.find(
   623→          (c) => frame >= c.startFrame && frame < c.startFrame + c.duration
   624→        );
   625→        if (cell) {
   626→          const idx = cells.indexOf(cell);
   627→          cells[idx] = { ...cell, drawingId };
   628→        } else {
   629→          // Crear nueva celda
   630→          cells.push({ id: genId("cell"), drawingId, startFrame: frame, duration: 1 });
   631→          cells.sort((a, b) => a.startFrame - b.startFrame);
   632→        }
   633→        return { ...l, cells };
   634→      });
   635→      return {
   636→        ...p,
   637→        layers,
   638→        drawings: { ...p.drawings, [drawingId]: drawing },
   639→      };
   640→    }));
   641→
   642→    return drawingId;
   643→  },

  1179→    // Restaurar el dataUrl anterior en el proyecto
  1180→    set((st) => {
  1181→      if (!st.project) return {};
  1182→      if (!st.project.drawings[entry.drawingId]) return {};
  1183→      const restored = {
  1184→        ...st.project,

  1187→          [entry.drawingId]: {
  1188→            ...st.project.drawings[entry.drawingId],

  1190→            updatedAt: Date.now(),
  1191→          },
  1192→        },
  1193→        dirty: true,

  1195→      };

  1198→        undoStack: st.undoStack.slice(0, -1),
  1199→        redoStack: [...st.redoStack, entry],
  1200→        skipReloadOnHistory: false,

  1202→    });
  1203→    // Pre-cachear la imagen restaurada

  1248→  startImagePlacement: (dataUrl, nativeW, nativeH) => {
  1249→    const s = get();
  1250→    if (!s.project) return;
  1251→    // Calcular tamaño inicial preservando aspect ratio:
  1252→    // - Si la imagen es más chica que el canvas, usar tamaño nativo
  1253→    // - Si es más grande, escalar para que entre (con margen del 10%)
  1254→    const cw = s.project.settings.width;
  1255→    const ch = s.project.settings.height;
  1256→    const margin = 0.1; // 10% de margen
  1257→    const maxW = cw * (1 - 2 * margin);
  1258→    const maxH = ch * (1 - 2 * margin);
  1259→    let w = nativeW;
  1260→    let h = nativeH;
  1261→    if (w > maxW || h > maxH) {
  1262→      const scale = Math.min(maxW / w, maxH / h);
  1263→      w = Math.round(w * scale);
  1264→      h = Math.round(h * scale);
  1265→    }
  1266→    // Centrar en el canvas
  1267→    const x = Math.round((cw - w) / 2);
  1268→    const y = Math.round((ch - h) / 2);
  1269→    set({
  1270→      imagePlacement: {
  1271→        dataUrl,
  1272→        x,
  1273→        y,
  1274→        width: w,
  1275→        height: h,
  1276→        nativeW,
  1277→        nativeH,
  1278→      },
  1279→    });
  1280→  },
  1281→
  1282→  updateImagePlacement: (patch) => {
  1283→    set((s) => ({
  1284→      imagePlacement: s.imagePlacement ? { ...s.imagePlacement, ...patch } : null,
  1285→    }));
  1286→  },
  1287→
  1288→  confirmImagePlacement: async () => {
  1289→    const s = get();
  1290→    if (!s.project || !s.imagePlacement) return;
  1291→    const placement = s.imagePlacement;
  1292→    // Renderizar la imagen al tamaño actual en el canvas del drawing activo
  1293→    // Crear nuevo drawing o usar el existente
  1294→    const layer = s.project.layers.find((l) => l.id === s.project!.currentLayerId);
  1295→    if (!layer || layer.locked || !layer.visible || layer.type === "audio") {
  1296→      set({ imagePlacement: null });
  1297→      return;
  1298→    }
  1299→
  1300→    // Cargar la imagen
  1301→    const img = await new Promise<HTMLImageElement | null>((resolve) => {
  1302→      const i = new Image();
  1303→      i.onload = () => resolve(i);
  1304→      i.onerror = () => resolve(null);
  1305→      i.src = placement.dataUrl;
  1306→    });
  1307→    if (!img) {
  1308→      set({ imagePlacement: null });
  1309→      return;
  1310→    }
  1311→
  1312→    // Crear un canvas temporal con el contenido actual del drawing + la imagen pegada
  1313→    const tmpCanvas = document.createElement("canvas");
  1314→    tmpCanvas.width = s.project.settings.width;
  1315→    tmpCanvas.height = s.project.settings.height;
  1316→    const tmpCtx = tmpCanvas.getContext("2d")!;
  1317→    // Si hay un drawing existente, dibujarlo primero
  1318→    const cell = layer.cells.find(
  1319→      (c) =>
  1320→        s.project!.currentFrame >= c.startFrame &&
  1321→        s.project!.currentFrame < c.startFrame + c.duration
  1322→    );
  1323→    if (cell?.drawingId) {
  1324→      const existing = s.project.drawings[cell.drawingId];
  1325→      if (existing) {
  1326→        const existingImg = await new Promise<HTMLImageElement | null>((resolve) => {
  1327→          const i = new Image();
  1328→          i.onload = () => resolve(i);
  1329→          i.onerror = () => resolve(null);
  1330→          i.src = existing.dataUrl;
  1331→        });
  1332→        if (existingImg) {
  1333→          tmpCtx.drawImage(existingImg, 0, 0, tmpCanvas.width, tmpCanvas.height);
  1334→        }
  1335→      }
  1336→    }
  1337→    // Dibujar la imagen colocada en su transformación final
  1338→    tmpCtx.drawImage(img, placement.x, placement.y, placement.width, placement.height);
  1339→    const finalDataUrl = tmpCanvas.toDataURL("image/png");
  1340→
  1341→    // Crear/actualizar el drawing con la imagen pegada
  1342→    const drawingId = cell?.drawingId ?? crypto.randomUUID();
  1343→    const now = Date.now();
  1344→    const prevDataUrl = cell?.drawingId ? s.project.drawings[cell.drawingId]?.dataUrl ?? "" : "";
  1345→
  1346→    // Pre-cachear la imagen final
  1347→    const { preloadImage } = await import("./useDrawingEngine");
  1348→    await preloadImage(drawingId, finalDataUrl);
  1349→
  1350→    set((st) => {
  1351→      if (!st.project) return {};
  1352→      const layers = st.project.layers.map((l) => {
  1353→        if (l.id !== layer.id) return l;
  1354→        const cells = [...l.cells];
  1355→        const existingCell = cells.find(
  1356→          (c) =>
  1357→            st.project!.currentFrame >= c.startFrame &&
  1358→            st.project!.currentFrame < c.startFrame + c.duration
  1359→        );
  1360→        if (existingCell) {
  1361→          const idx = cells.indexOf(existingCell);
  1362→          cells[idx] = { ...existingCell, drawingId };
  1363→        } else {
  1364→          cells.push({
  1365→            id: crypto.randomUUID(),
  1366→            drawingId,
  1367→            startFrame: st.project.currentFrame,
  1368→            duration: 1,
  1369→          });
  1370→          cells.sort((a, b) => a.startFrame - b.startFrame);
  1371→        }
  1372→        return { ...l, cells };
  1373→      });
  1374→      return {
  1375→        project: {
  1376→          ...st.project,
  1377→          layers,
  1378→          drawings: {
  1379→            ...st.project.drawings,
  1380→            [drawingId]: {
  1381→              id: drawingId,
  1382→              name: "Imagen importada",
  1383→              dataUrl: finalDataUrl,
  1384→              width: st.project.settings.width,
  1385→              height: st.project.settings.height,
  1386→              createdAt: now,
  1387→              updatedAt: now,
  1388→            },
  1389→          },
  1390→          dirty: true,
  1391→          updatedAt: now,
  1392→        },
  1393→        imagePlacement: null,
  1394→      };
  1395→    });
  1396→
  1397→    // Registrar en el stack de undo
  1398→    if (prevDataUrl && prevDataUrl !== finalDataUrl) {
  1399→      get().pushHistory(drawingId, prevDataUrl, finalDataUrl);
  1400→    }
  1401→  },
  1402→
  1403→  cancelImagePlacement: () => {
  1404→    set({ imagePlacement: null });
  1405→  },
  1406→
  1407→  triggerAutosave: async () => {
